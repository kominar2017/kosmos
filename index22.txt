function sliderPosToKm(pos) {
  // Логарифмическая шкала: pos 0 → 1 км, pos 1000 → reviewMaxKm
  if (reviewMaxKm <= 1) return 1;
  const logMax = Math.log10(reviewMaxKm);
  const logKm = (pos / 1000) * logMax;
  return Math.pow(10, logKm);
}

function kmToSliderPos(km) {
  if (km < 1) km = 1;
  if (reviewMaxKm <= 1) return 0;
  const logMax = Math.log10(reviewMaxKm);
  const logKm = Math.log10(km);
  const pos = (logKm / logMax) * 1000;
  return Math.max(0, Math.min(1000, pos));
}

function formatDistance(n) {
  if (n < 0n) n = 0n;
  const s = n.toString();
  if (s.length <= 15) return s.replace(/\B(?=(\d{3})+(?!\d))/g, ' ');
  const exp = s.length - 1;
  return s[0] + '.' + s.substring(1, 3) + ' × 10^' + exp;
}

function formatKmNumber(km) {
  if (km < 1) return '1 км';
  if (km < 1000000) return Math.round(km).toLocaleString('ru-RU') + ' км';
  const exp = Math.floor(Math.log10(km));
  const mantissa = km / Math.pow(10, exp);
  return mantissa.toFixed(2) + ' × 10^' + exp + ' км';
}

const LAYERS = [
  { speedMul: 0.35, sizeMul: 0.6, alphaMul: 0.55, count: 200 },

const stars = [];

  for (let i = 0; i < LAYERS[li].count; i++) {
    stars.push({
      x: (Math.random() - 0.5) * W * 2.0,

function drawStars() {
  const km = getCurrentKm();
  const starAlpha = 0.35 + Math.min(0.65, km / 200);

  for (const star of stars) {
    const layer = LAYERS[star.layer];
    const effectiveSpeed = starsSpeed * layer.speedMul;
    const oldZ = star.z;
    star.z -= effectiveSpeed;
    if (star.z <= 0) {
      star.x = (Math.random() - 0.5) * W * 2.0;
      star.y = (Math.random() - 0.5) * H * 2.0;
      star.z = 1000;
      continue;
    }
    const k = 300 / star.z;
    const px = star.x * k + W / 2;
    const py = star.y * k + H / 2;
    const kOld = 300 / oldZ;
    const pxOld = star.x * kOld + W / 2;
    const pyOld = star.y * kOld + H / 2;
    if (px < -50 || px > W + 50 || py < -50 || py > H + 50) continue;
    const size = (1 - star.z / 1000) * 2.4 * layer.sizeMul;
    const alpha = (1 - star.z / 1000) * starAlpha * layer.alphaMul;
    const speedRatio = Math.max(0, Math.min(1, (starsSpeed - 3) / 8));
    if (speedRatio > 0.1) {
      const len = Math.abs(px - pxOld) + Math.abs(py - pyOld);
      if (len > 1.5) {
        ctx.beginPath();
        ctx.moveTo(pxOld, pyOld);
        ctx.lineTo(px, py);
        ctx.strokeStyle = `rgba(200, 220, 255, ${alpha * (0.6 + speedRatio * 0.4)})`;
        ctx.lineWidth = Math.max(0.4, size * (0.5 + speedRatio * 0.5));
        ctx.stroke();
        continue;
      }
    }
    ctx.beginPath();
    ctx.arc(px, py, size, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(200, 220, 255, ${alpha})`;
    ctx.fill();
  }
}

function drawPlanetSphere(x, y, r, colors, atmosphereColor, atmosphereStrength) {
  if (atmosphereStrength > 0 && atmosphereColor) {
    const [ar, ag, ab] = atmosphereColor;
    const glow = ctx.createRadialGradient(x, y, r * 0.9, x, y, r * 1.4);
    glow.addColorStop(0, `rgba(${ar}, ${ag}, ${ab}, ${0.45 * atmosphereStrength})`);
    glow.addColorStop(0.5, `rgba(${ar}, ${ag}, ${ab}, ${0.2 * atmosphereStrength})`);
    glow.addColorStop(1, `rgba(${ar}, ${ag}, ${ab}, 0)`);
    ctx.fillStyle = glow;
    ctx.beginPath(); ctx.arc(x, y, r * 1.4, 0, Math.PI * 2); ctx.fill();
  }

  const lightX = x + r * 0.4;
  const lightY = y - r * 0.5;

  const baseGrad = ctx.createRadialGradient(lightX, lightY, r * 0.05, x, y, r * 1.05);
  for (let i = 0; i < colors.length; i++) {
    baseGrad.addColorStop(i / (colors.length - 1), colors[i]);
  }
  ctx.fillStyle = baseGrad;
  ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fill();

  const dark = ctx.createRadialGradient(
    x - r * 0.6, y + r * 0.7, r * 0.1,
    x - r * 0.3, y + r * 0.4, r * 1.2
  );
  dark.addColorStop(0, 'rgba(0, 0, 0, 0)');
  dark.addColorStop(0.5, 'rgba(0, 0, 0, 0.15)');
  dark.addColorStop(1, 'rgba(0, 0, 0, 0.5)');
  ctx.fillStyle = dark;
  ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fill();

  const highlight = ctx.createRadialGradient(lightX, lightY, 0, lightX, lightY, r * 0.6);
  highlight.addColorStop(0, 'rgba(255, 255, 255, 0.2)');
  highlight.addColorStop(1, 'rgba(255, 255, 255, 0)');
  ctx.fillStyle = highlight;
  ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fill();
}

function projectSphere(lonDeg, latDeg, cx, cy, r) {
  const lon = lonDeg * Math.PI / 180;
  const lat = latDeg * Math.PI / 180;
  const cosLat = Math.cos(lat);
  const sinLat = Math.sin(lat);
  const cosLon = Math.cos(lon);
  const sinLon = Math.sin(lon);
  if (cosLon <= 0.05) return null;
  return { x: cx + r * sinLon * cosLat, y: cy - r * sinLat, depth: cosLon };
}

function smoothPolygon(points) {
  if (points.length < 3) return;
  ctx.beginPath();
  const n = points.length;
  const startX = (points[n-1].x + points[0].x) / 2;
  const startY = (points[n-1].y + points[0].y) / 2;
  ctx.moveTo(startX, startY);
  for (let i = 0; i < n; i++) {
    const p = points[i];
    const next = points[(i + 1) % n];
    const mx = (p.x + next.x) / 2;
    const my = (p.y + next.y) / 2;
    ctx.quadraticCurveTo(p.x, p.y, mx, my);
  }
  ctx.closePath();
}

function drawEarth(km) {

  const cx = W / 2, cy = H * 0.88;

    ctx.lineWidth = Math.max(0.5, r * 0.008);
    ctx.stroke();
    ctx.fillStyle = cont.color;
    ctx.fill();

    ctx.save();
    ctx.clip();
    const contLight = ctx.createRadialGradient(lightX, lightY, 0, lightX, lightY, r * 1.2);
    contLight.addColorStop(0, 'rgba(255, 240, 200, 0.2)');
    contLight.addColorStop(0.5, 'rgba(255, 240, 200, 0)');
    ctx.fillStyle = contLight;
    ctx.fillRect(cx - r, cy - r, r * 2, r * 2);
    ctx.restore();
  }

  ctx.beginPath(); ctx.ellipse(cx, cy + r * 1.0, r * 1.0, r * 0.65, 0, 0, Math.PI * 2); ctx.fill();

  const cloudRotation = globalTime * 5;
  const cloudSpots = [

  const specular = ctx.createRadialGradient(lightX, lightY, 0, lightX, lightY, r * 0.55);

  ctx.fillStyle = haze;
  ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fill();

  const dark = ctx.createRadialGradient(
    cx - r * 0.55, cy + r * 0.65, r * 0.2,
    cx - r * 0.4, cy + r * 0.5, r * 1.15
  );
  dark.addColorStop(0, 'rgba(0, 0, 0, 0)');

  ctx.lineWidth = Math.max(1, r * 0.008);
  ctx.beginPath(); ctx.arc(cx, cy, r * 1.005, 0, Math.PI * 2); ctx.stroke();
}

   ========================================================= */
function drawBody(body, km) {
  const objKm = Number(body.distance);
  const ratio = km / objKm;
  if (ratio < 0.3 || ratio > 1.9) return;
  const delta = km - objKm;
  const absDelta = Math.abs(delta);
  const visibility = objKm * 0.9;
  if (absDelta > visibility) return;
  const proximity = 1 - absDelta / visibility;
  const baseSize = Math.min(W, H) * body.size;
  const size = baseSize * (0.1 + proximity * 0.9);
  const r = size * 0.5;
  const side = body.pathSide || 0;
  const offset = body.pathOffset || 0;
  const xFactor = side * W * offset * (0.6 + (1 - proximity) * 0.6);
  const x = W / 2 + xFactor;
  const y = H / 2 + (delta / visibility) * H * 0.45;
  if (y + r * 1.5 < -200 || y - r * 1.5 > H + 200) return;
  if (size < 3) return;
  const opacity = Math.min(1, (size - 3) / 12);
  ctx.globalAlpha = opacity;

  if (body.type === 'moon')        drawMoon(x, y, r, body);
  else if (body.type === 'mars')   drawMars(x, y, r, body);
  else if (body.type === 'venus')  drawVenus(x, y, r, body);
  else if (body.type === 'mercury')drawMercury(x, y, r, body);
  else if (body.type === 'jupiter')drawJupiter(x, y, r, body);
  else if (body.type === 'saturn') drawSaturn(x, y, r, body);
  else if (body.type === 'uranus') drawUranus(x, y, r, body);
  else if (body.type === 'neptune')drawNeptune(x, y, r, body);
  else if (body.type === 'pluto')  drawPluto(x, y, r, body);
  else if (body.type === 'dwarf')  drawDwarf(x, y, r, body);
  else if (body.type === 'station')drawStation(x, y, r);
  else if (body.type === 'hubble') drawHubble(x, y, r);
  else if (body.type === 'starlink')drawStarlink(x, y, r);
  else if (body.type === 'satellite')drawSatellite(x, y, r);
  else if (body.type === 'point')  drawLagrangePoint(x, y, r);
  else if (body.type === 'galilean')drawGalilean(x, y, r);
  else if (body.type === 'titan')  drawTitan(x, y, r, body);
  else if (body.type === 'probe')  drawProbe(x, y, r);
  else if (body.type === 'star')   drawStarBody(x, y, r);
  else if (body.type === 'supergiant')drawSupergiant(x, y, r);

  ctx.globalAlpha = 1;
}

function drawMoon(x, y, r, body) {
  drawPlanetSphere(x, y, r,
    ['#f4f4f8', '#dcdce4', '#b8b8c4', '#80808c', '#3a3a48'],
    [200, 210, 230], 0.12);
  if (r > 12) {
    ctx.save();
    ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.clip();
    const rot = (body.rotationSpeed || 0.15) * globalTime;
    for (const m of MOON_MARIA) {
      const p = projectSphere(m.lon + rot, m.lat, x, y, r);
      if (!p) continue;
      const mr = r * m.r * p.depth;
      if (mr < 2) continue;
      const seaGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, mr);
      seaGrad.addColorStop(0, 'rgba(60, 62, 75, 0.5)');
      seaGrad.addColorStop(0.7, 'rgba(75, 78, 92, 0.3)');
      seaGrad.addColorStop(1, 'rgba(75, 78, 92, 0)');
      ctx.fillStyle = seaGrad;
      ctx.beginPath(); ctx.arc(p.x, p.y, mr, 0, Math.PI * 2); ctx.fill();
    }
    for (const c of MOON_CRATERS) {
      const p = projectSphere(c.lon + rot, c.lat, x, y, r);
      if (!p) continue;
      const cr = r * c.r * p.depth;
      if (cr < 1.5) continue;
      const lx = Math.cos(-Math.PI * 0.7) * cr * 0.3;
      const ly = Math.sin(-Math.PI * 0.7) * cr * 0.3;
      ctx.fillStyle = `rgba(50, 52, 65, ${0.55 * p.depth})`;
      ctx.beginPath(); ctx.arc(p.x - lx, p.y - ly, cr * 0.9, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = `rgba(220, 225, 235, ${0.5 * p.depth})`;
      ctx.beginPath(); ctx.arc(p.x + lx, p.y + ly, cr * 0.85, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = `rgba(45, 48, 60, ${0.5 * p.depth})`;
      ctx.beginPath(); ctx.arc(p.x, p.y, cr * 0.65, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = `rgba(180, 185, 200, ${0.3 * p.depth})`;
      ctx.beginPath(); ctx.arc(p.x + lx * 0.3, p.y + ly * 0.3, cr * 0.25, 0, Math.PI * 2); ctx.fill();
    }
    ctx.restore();
  }
}

function drawBelt(belt, km) {
  for (const ast of belt) {
    const objKm = ast.distance;
    const ratio = km / objKm;
    if (ratio < 0.3 || ratio > 1.9) continue;
    const delta = km - objKm;
    const absDelta = Math.abs(delta);
    const visibility = objKm * 0.85;
    if (absDelta > visibility) continue;
    const proximity = 1 - absDelta / visibility;
    const xOff = ast.xOff * W * 0.55 * (1 - proximity * 0.5);
    const x = W / 2 + xOff;
    const yOff = ast.yOff * H * 0.55 * (1 - proximity * 0.5);
    const y = H / 2 + yOff + (delta / visibility) * H * 0.4;
    if (x < -20 || x > W + 20 || y < -20 || y > H + 20) continue;
    const size = ast.size * (1 + proximity * 2.5) * 0.8;
    const alpha = ast.alpha * (0.3 + proximity * 0.7);
    const rotation = ast.rot + globalTime * ast.rotSpeed;
    ctx.beginPath();
    const n = ast.verts.length;
    for (let i = 0; i < n; i++) {
      const angle = ast.verts[i] + rotation;
      const rad = size * (0.8 + Math.sin(i * 1.7 + ast.rot) * 0.2);
      const px = x + Math.cos(angle) * rad;
      const py = y + Math.sin(angle) * rad;
      if (i === 0) ctx.moveTo(px, py);
      else ctx.lineTo(px, py);
    }
    ctx.closePath();
    const grad = ctx.createRadialGradient(x - size * 0.3, y - size * 0.3, 0, x, y, size);
    grad.addColorStop(0, `rgba(180, 170, 150, ${alpha})`);
    grad.addColorStop(0.6, `rgba(120, 110, 95, ${alpha * 0.9})`);
    grad.addColorStop(1, `rgba(60, 55, 50, ${alpha * 0.8})`);
    ctx.fillStyle = grad;
    ctx.fill();
  }
}

function kmToScaleX(kmVal, x1, barW) {
  if (kmVal < 1) kmVal = 1;
  const log = Math.log10(kmVal);
  const ratio = Math.min(1, Math.max(0, log / LOG_MAX));
  return x1 + ratio * barW;
}

function drawScale() {
  if (reviewMode) return;   // в режиме просмотра шкалу не рисуем
  const km = Number(distance);
  const barY = H - 46;
  const x1 = W * 0.06;
  const x2 = W * 0.94;
  const barW = x2 - x1;

  ctx.strokeStyle = 'rgba(140, 180, 240, 0.15)';
  ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(x1, barY); ctx.lineTo(x2, barY); ctx.stroke();

  const progressX = kmToScaleX(Math.max(km, 1), x1, barW);
  const grad = ctx.createLinearGradient(x1, barY, progressX, barY);
  grad.addColorStop(0, 'rgba(140, 180, 240, 0.35)');
  grad.addColorStop(1, 'rgba(160, 210, 255, 0.75)');
  ctx.strokeStyle = grad;
  ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(x1, barY); ctx.lineTo(progressX, barY); ctx.stroke();

  ctx.textAlign = 'center';
  ctx.font = '300 9px "Space Mono", monospace';
  for (const m of SCALE_MARKS) {
    const x = kmToScaleX(m.km, x1, barW);
    const passed = km >= m.km;
    ctx.fillStyle = passed ? 'rgba(180, 220, 255, 0.85)' : 'rgba(140, 180, 240, 0.35)';
    ctx.beginPath(); ctx.arc(x, barY, passed ? 2.5 : 2, 0, Math.PI * 2); ctx.fill();
    const yOff = m.level * 10;
    ctx.strokeStyle = passed ? 'rgba(180, 220, 255, 0.35)' : 'rgba(140, 180, 240, 0.12)';
    ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(x, barY); ctx.lineTo(x, barY + yOff + 3); ctx.stroke();
    ctx.fillStyle = passed ? 'rgba(200, 225, 255, 0.7)' : 'rgba(160, 190, 230, 0.3)';
    ctx.fillText(m.label, x, barY + yOff + 14);
  }

  ctx.shadowColor = 'rgba(160, 220, 255, 0.9)';
  ctx.shadowBlur = 12;
  ctx.fillStyle = '#d0e8ff';
  ctx.beginPath(); ctx.arc(progressX, barY, 2.5, 0, Math.PI * 2); ctx.fill();
  ctx.shadowBlur = 0;
}

/* =========================================================
   АНИМАЦИЯ
   ========================================================= */
function animate(now) {
  const dt = Math.min((now - lastFrameTime) / 1000, 0.05);
  lastFrameTime = now;
  globalTime += dt;

  // Логика просмотра
  if (reviewMode && reviewPlaying) {
    reviewSliderPos += PLAY_SPEED * dt;
    if (reviewSliderPos >= 1000) {
      reviewSliderPos = 1000;
      reviewPlaying = false;
      document.getElementById('reviewPlay').textContent = '▶';
    }
    document.getElementById('reviewSlider').value = reviewSliderPos;
    viewKm = sliderPosToKm(reviewSliderPos);
    document.getElementById('reviewKmLabel').textContent = formatKmNumber(viewKm);
    distanceEl.textContent = formatDistance(BigInt(Math.round(viewKm)));
  }

  // Основной разгон — только в обычном режиме
  if (!reviewMode) {
    impulse *= Math.pow(0.5, dt / IMPULSE_HALFLIFE);
    if (impulse < 0.05) impulse = 0;

    const km = Number(distance);
    const starFactor = 0.4 + Math.min(0.6, km / 200);
    targetSpeed = BASE_SPEED * starFactor + impulse * 0.9;
    if (targetSpeed > MAX_SPEED) targetSpeed = MAX_SPEED;
    starsSpeed += (targetSpeed - starsSpeed) * 0.12;

    const isBoost = impulse >= BOOST_THRESHOLD;
    if (isBoost !== boostActive) {
      boostActive = isBoost;
      counterEl.classList.toggle('boost', isBoost);
    }
  } else {
    // В режиме просмотра — медленный дрейф звёзд
    targetSpeed = 1.5;
    starsSpeed += (targetSpeed - starsSpeed) * 0.05;
    counterEl.classList.remove('boost');
  }

  ctx.fillStyle = 'rgba(0, 0, 5, 0.35)';
  ctx.fillRect(0, 0, W, H);

  const km = getCurrentKm();

  drawNebulae();
  drawStars();
  drawSun(km);
  drawEarth(km);
  drawBelt(asteroidBelt, km);
  drawBelt(kuiperBelt, km);
  for (const body of BODIES) drawBody(body, km);
  drawScale();

  requestAnimationFrame(animate);
}
requestAnimationFrame((t) => { lastFrameTime = t; animate(t); });

function handleClick(e) {
  if (e && e.target && (e.target.tagName === 'INPUT' || e.target.tagName === 'BUTTON')) return;
  if (reviewMode) return;   // в режиме просмотра клики не считаем

  distance += 10n;
  impulse += IMPULSE_PER_CLICK;
  if (impulse > IMPULSE_MAX) impulse = IMPULSE_MAX;
  hintEl.style.opacity = '0';
  distanceEl.textContent = formatDistance(distance);
  checkMilestones();
  save();
}
document.body.addEventListener('click', handleClick);

function checkMilestones() {
  if (nextMilestone >= MILESTONES.length) return;
  const m = MILESTONES[nextMilestone];
  if (distance >= m.d) {
    showMilestone(m.text);
    nextMilestone++;
    checkMilestones();
  }
}

let milestoneTimer = null;
function showMilestone(text) {
  milestoneEl.textContent = text;
  milestoneEl.classList.add('show');
  clearTimeout(milestoneTimer);
  milestoneTimer = setTimeout(() => milestoneEl.classList.remove('show'), 7000);
}

function save() {
  try {
    localStorage.setItem('space_distance', distance.toString());
    localStorage.setItem('space_milestone', nextMilestone.toString());
  } catch (e) {}
}

function load() {
  try {
    const d = localStorage.getItem('space_distance');
    const m = localStorage.getItem('space_milestone');
    if (d) distance = BigInt(d);
    if (m) nextMilestone = parseInt(m, 10);
    distanceEl.textContent = formatDistance(distance);
    if (distance > 0n) hintEl.style.opacity = '0';
  } catch (e) {}
}
load();

const jumpInput = document.getElementById('jumpInput');

function jumpToKm(str) {
  const clean = String(str).trim().replace(/[\s,]/g, '');
  const km = Number(clean);
  if (!isFinite(km) || km < 0) return;
  distance = BigInt(Math.round(km));
  distanceEl.textContent = formatDistance(distance);
  while (nextMilestone < MILESTONES.length && distance >= MILESTONES[nextMilestone].d) {
    nextMilestone++;
  }
  save();
}

jumpInput.addEventListener('keydown', (e) => {
  if (e.key === 'Enter') {
    e.preventDefault();
    jumpToKm(jumpInput.value);
    jumpInput.blur();
  }
});
jumpBtn.addEventListener('click', (e) => {
  e.preventDefault();
  e.stopPropagation();
  jumpToKm(jumpInput.value);
});
